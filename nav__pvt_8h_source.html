169static inline bool NAV_PVT_decode(
170 const volatile uint8_t *rx_buffer,
171 NAV_PVT_payload_t *nav_pvt
172) {
173 ubx_header_t header; // create copy to avoid alignment issues
174 memcpy(&header, (const void *)rx_buffer, sizeof(ubx_header_t));
175
176 if (!UBX_is_matching_header(&header, &NAV_PVT_HEADER)) {
177 return false;
178 }
179
180 if (!UBX_is_valid_checksum((const uint8_t *)rx_buffer, NAV_PVT_HEADER.payload_length)) {
181 return false;
182 }
183
184 memcpy(nav_pvt, (const void *)(rx_buffer + UBX_HEADER_LENGTH), sizeof(NAV_PVT_payload_t));
185
186 return true;
187}
