16// This board's external oscillator input
17// HSI16 (High Speed Internal, 16 MHz) is fixed by hardware
18// HSE (High Speed External) depends on PCB/what is connected wired to the OSC_IN pin
19// and PER only uses 16 MHZ HSE fed bypass style
20static constexpr uint32_t RCC_PRIV_HSI_CLOCK_RATE_HZ = 16'000'000U;
21static constexpr uint32_t RCC_PRIV_HSE_CLOCK_RATE_HZ = 16'000'000U;
22
23// 170 MHz PLL
24// VCO input = HSI16 or HSE / PLLM (must be 2.66-16 MHz)
25// VCO output = VCO input * PLLN (must be 96-344 MHz)
26// PLL output = VCO output / PLLR (-> SYSCLK)
27static constexpr uint32_t RCC_PRIV_PLL_M = 4U;
28static constexpr uint32_t RCC_PRIV_PLL_N = 85U;
29static constexpr uint32_t RCC_PRIV_PLL_R = 2U;
30
31static constexpr uint32_t RCC_PRIV_PLL_VCO_INPUT_HZ = RCC_PRIV_HSI_CLOCK_RATE_HZ / RCC_PRIV_PLL_M;
32static constexpr uint32_t RCC_PRIV_PLL_VCO_OUTPUT_HZ = RCC_PRIV_PLL_VCO_INPUT_HZ * RCC_PRIV_PLL_N;
33static constexpr uint32_t RCC_PRIV_PLL_OUTPUT_HZ = RCC_PRIV_PLL_VCO_OUTPUT_HZ / RCC_PRIV_PLL_R;
34
35// Compile time check that chosen M/N/R choice is legal and results in a perfect 170 MHZ
36static_assert(RCC_PRIV_HSI_CLOCK_RATE_HZ == RCC_PRIV_HSE_CLOCK_RATE_HZ, "PLL config assumes HSI and HSE");
37static_assert(RCC_PRIV_PLL_VCO_INPUT_HZ >= 2'660'000U && RCC_PRIV_PLL_VCO_INPUT_HZ <= 16'000'000U, "PLL VCO input out of valid range");
38static_assert(RCC_PRIV_PLL_VCO_OUTPUT_HZ >= 96'000'000U && RCC_PRIV_PLL_VCO_OUTPUT_HZ <= 344'000'000U, "PLL VCO output out of valid range");
39static_assert(RCC_PRIV_PLL_OUTPUT_HZ == 170'000'000U, "PLL output must be exactly 170 MHz");
